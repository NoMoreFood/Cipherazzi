cmake_minimum_required(VERSION 3.28)
project(StaticTlsFixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)

set(TLS_INCLUDE_DIRECTORY "" CACHE PATH "TLS library headers")
set(TLS_SSL_LIBRARY "" CACHE FILEPATH "Static TLS library")
set(TLS_CRYPTO_LIBRARY "" CACHE FILEPATH "Static cryptography library")
set(TLS_SYMBOL_NAME "StaticTlsSymbols" CACHE STRING "Local PDB filename")
option(TLS_SHARED_FIXTURE "Load the statically linked TLS implementation from a DLL" OFF)

if(TLS_SHARED_FIXTURE)
    add_library(StaticTlsLibrary SHARED Client.cpp)
    target_compile_definitions(StaticTlsLibrary PRIVATE STATIC_TLS_DLL)
    set(TLS_TARGET StaticTlsLibrary)
    add_executable(StaticTlsLoader Loader.cpp)
    target_compile_definitions(StaticTlsLoader PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
else()
    add_executable(StaticTlsClient Client.cpp)
    set(TLS_TARGET StaticTlsClient)
endif()

set_target_properties(${TLS_TARGET} PROPERTIES PDB_NAME "${TLS_SYMBOL_NAME}")
target_include_directories(${TLS_TARGET} PRIVATE "${TLS_INCLUDE_DIRECTORY}")
target_compile_definitions(${TLS_TARGET} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(${TLS_TARGET} PRIVATE "${TLS_SSL_LIBRARY}" "${TLS_CRYPTO_LIBRARY}" ws2_32 crypt32)
